score-average.js
const scores = [87, 70, 90, 95, 60];
const total = scores.reduce((sum, score) => sum + score, 0);
const average = total / 5;

if (average >= 90) {
  console.log("A · Excellent!");
} else if (average >= 80) {
  console.log("B · Great work!");
} else if (average >= 70) {
  console.log("C · Good effort!");
} else if (average >= 60) {
  console.log("D · Keep practicing.");
} else {
  console.log("F · Try again.");
}
